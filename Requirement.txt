Database Query 1:
-- 1. Tabel Users (Menyimpan data akun: Customer, Guide, atau Admin)
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    phone VARCHAR(20),
    city VARCHAR(50),
    address TEXT,
    shoe_size VARCHAR(5),
    hiking_level VARCHAR(20) DEFAULT 'Pemula',
    role VARCHAR(20) DEFAULT 'customer', -- 'customer', 'guide', 'admin'
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. Tabel Products (Menyimpan alat camping untuk Beli dan Sewa)
CREATE TABLE products (
    id SERIAL PRIMARY KEY,
    name VARCHAR(150) NOT NULL,
    category VARCHAR(50) NOT NULL, -- Contoh: 'Tenda', 'Tas', 'Masak', 'Tidur'
    description TEXT,
    price DECIMAL(12, 2) NOT NULL, -- Harga Beli
    rent_price_per_day DECIMAL(12, 2), -- Harga Sewa per hari
    stock INT DEFAULT 0,
    image_url TEXT,
    is_rentable BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 3. Tabel Guides (Menyimpan profil pemandu / porter profesional)
CREATE TABLE guides (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id) ON DELETE CASCADE,
    name VARCHAR(100) NOT NULL,
    mountain_region VARCHAR(150) NOT NULL, -- Contoh: 'Gunung Gede Pangrango', 'Semeru'
    price_per_day DECIMAL(12, 2) NOT NULL,
    rating DECIMAL(3, 2) DEFAULT 5.00,
    total_trips INT DEFAULT 0,
    experience_years INT DEFAULT 1,
    verified BOOLEAN DEFAULT TRUE,
    skills TEXT[], -- Array keahlian, misal: ARRAY['First Aid', 'Memasak', 'Fotografi']
    avatar_url TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 4. Tabel Guide Bookings (Menyimpan riwayat booking guide dari user)
CREATE TABLE guide_bookings (
    id SERIAL PRIMARY KEY,
    customer_id INT REFERENCES users(id) ON DELETE CASCADE,
    guide_id INT REFERENCES guides(id) ON DELETE CASCADE,
    destination VARCHAR(150) NOT NULL,
    start_date DATE NOT NULL,
    total_days INT NOT NULL,
    include_porter BOOLEAN DEFAULT FALSE,
    include_photo BOOLEAN DEFAULT FALSE,
    total_price DECIMAL(12, 2) NOT NULL,
    status VARCHAR(30) DEFAULT 'Menunggu Konfirmasi', -- 'Menunggu Konfirmasi', 'Dikonfirmasi', 'Selesai', 'Dibatalkan'
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 5. Tabel Orders / Transaksi Pembelian & Penyewaan Produk
CREATE TABLE orders (
    id SERIAL PRIMARY KEY,
    customer_id INT REFERENCES users(id) ON DELETE CASCADE,
    order_type VARCHAR(20) NOT NULL, -- 'beli' atau 'sewa'
    total_price DECIMAL(12, 2) NOT NULL,
    status VARCHAR(30) DEFAULT 'Diproses', -- 'Diproses', 'Dikirim', 'Selesai', 'Dibatalkan'
    rental_start_date DATE, -- Diisi jika transaksinya sewa
    rental_end_date DATE,   -- Diisi jika transaksinya sewa
    shipping_address TEXT,
    receipt_number VARCHAR(50), -- Resi pengiriman
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 6. Tabel Order Items (Detail produk apa saja yang dibeli/disewa dalam 1 order)
CREATE TABLE order_items (
    id SERIAL PRIMARY KEY,
    order_id INT REFERENCES orders(id) ON DELETE CASCADE,
    product_id INT REFERENCES products(id) ON DELETE SET NULL,
    quantity INT NOT NULL,
    price_at_time DECIMAL(12, 2) NOT NULL
);

-- Tabel Utama Paket Bundling
CREATE TABLE bundles (
    id SERIAL PRIMARY KEY,
    bundle_key VARCHAR(50) UNIQUE NOT NULL, -- Contoh: 'pemula', 'summit'
    label VARCHAR(100) NOT NULL,            -- Contoh: 'Pemula Weekend'
    icon VARCHAR(50) DEFAULT 'fa-campground',
    color VARCHAR(20) DEFAULT '#15803d',    -- Warna tema kartu (Hex)
    badge VARCHAR(50),                      -- Contoh: 'Terlaris'
    name VARCHAR(150) NOT NULL,             -- Contoh: 'Paket Pemula Weekend'
    tag TEXT,                               -- Deskripsi singkat
    meta_tags TEXT[],                       -- Contoh: ARRAY['2 orang', '2D1N']
    discount_percentage INT DEFAULT 10,     -- Diskon dalam persen
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Tabel Item di dalam Paket Bundling (Relasi ke produk)
CREATE TABLE bundle_items (
    id SERIAL PRIMARY KEY,
    bundle_id INT REFERENCES bundles(id) ON DELETE CASCADE,
    icon VARCHAR(50) DEFAULT 'fa-tent',
    item_name VARCHAR(150) NOT NULL,        -- Contoh: 'Tenda Dome 2P'
    item_desc TEXT,                         -- Contoh: 'Double layer, anti air'
    price DECIMAL(12, 2) NOT NULL           -- Harga normal item
);

-- 1. Tabel utama paket bundling
CREATE TABLE IF NOT EXISTS bundles (
    id SERIAL PRIMARY KEY,
    bundle_key VARCHAR(50) UNIQUE NOT NULL, -- contoh: 'camping-ceria', 'pro-climb'
    label VARCHAR(100) NOT NULL,            -- contoh: 'PAKET HEMAT'
    name VARCHAR(150) NOT NULL,             -- contoh: 'Paket Camping Ceria 4 Orang'
    tag VARCHAR(100),                       -- contoh: 'Cocok untuk keluarga / 4 orang'
    badge VARCHAR(50),                      -- contoh: 'Hemat 20%'
    discount_percentage INTEGER DEFAULT 0,  -- contoh: 20
    icon VARCHAR(50) DEFAULT 'fa-campground',
    color VARCHAR(50) DEFAULT 'orange',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. Tabel relasi item di dalam bundling
CREATE TABLE IF NOT EXISTS bundle_items (
    id SERIAL PRIMARY KEY,
    bundle_id INTEGER REFERENCES bundles(id) ON DELETE CASCADE,
    item_name VARCHAR(150) NOT NULL,
    item_desc VARCHAR(255),
    price NUMERIC(12,2) NOT NULL,
    icon VARCHAR(50) DEFAULT 'fa-check'
);
